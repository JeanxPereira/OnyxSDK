cmake_minimum_required(VERSION 3.28)

set(ENV{VSLANG} 1033)
project(OnyxSDK VERSION 2.0.0 LANGUAGES CXX C)
set(CMAKE_CXX_SCAN_FOR_MODULES OFF)
cmake_policy(VERSION 3.28...4.2)

include(CMakeDependentOption)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

option(ONYX_WITH_GPU "Build Onyx::Gpu and Onyx::Render" ON)
cmake_dependent_option(ONYX_WITH_UI "Build Onyx::Ui" ON "ONYX_WITH_GPU" OFF)
cmake_dependent_option(ONYX_WITH_SHELL "Build Onyx::Shell" ON "ONYX_WITH_UI" OFF)
cmake_dependent_option(ONYX_WITH_MEDIA "Build Onyx::Media" ON "ONYX_WITH_UI" OFF)
option(ONYX_BUILD_TESTS "Build the test executables" ${PROJECT_IS_TOP_LEVEL})
option(ONYX_BUILD_EXAMPLES "Build the examples" ${PROJECT_IS_TOP_LEVEL})
option(ONYX_BUILD_TOOLS "Build the tools" ${PROJECT_IS_TOP_LEVEL})

message(STATUS "Onyx components: GPU=${ONYX_WITH_GPU} UI=${ONYX_WITH_UI} SHELL=${ONYX_WITH_SHELL} MEDIA=${ONYX_WITH_MEDIA}")

include(cmake/Warnings.cmake)
include(cmake/Layer.cmake)
include(cmake/Version.cmake)

add_subdirectory(Source/Core)
add_subdirectory(Source/Cli)
add_subdirectory(Source/TestKit)
add_subdirectory(Source/Exchange)
if(ONYX_WITH_GPU)
    add_subdirectory(Source/Gpu)
    add_subdirectory(Source/Render)
endif()
if(ONYX_WITH_UI)
    add_subdirectory(Source/Ui)
endif()
if(ONYX_WITH_SHELL)
    add_subdirectory(Source/Shell)
endif()
if(ONYX_WITH_MEDIA)
    add_subdirectory(Source/Media)
endif()

enable_testing()
if(ONYX_BUILD_TESTS)
    add_subdirectory(Tests)
endif()
if(ONYX_BUILD_TESTS OR ONYX_BUILD_EXAMPLES)
    add_subdirectory(Examples/OnyxBox)
endif()
if(ONYX_BUILD_EXAMPLES)
    add_subdirectory(Examples)
endif()
